cmake_minimum_required(VERSION 3.20)
project(ExoticCoreRuntime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(sqlite_amalgamation URL https://www.sqlite.org/2026/sqlite-amalgamation-3530400.zip DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_GetProperties(sqlite_amalgamation)
if(NOT sqlite_amalgamation_POPULATED)
  FetchContent_Populate(sqlite_amalgamation)
endif()
add_library(exotic_sqlite STATIC ${sqlite_amalgamation_SOURCE_DIR}/sqlite3.c)
target_include_directories(exotic_sqlite PUBLIC ${sqlite_amalgamation_SOURCE_DIR})
target_compile_definitions(exotic_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION=1)
add_library(exotic_core src/runtime.cpp src/free_agent.cpp src/free_agent_service.cpp src/agent_store.cpp src/idle_cognition.cpp src/operator_sessions.cpp src/self_model.cpp src/metacognition.cpp src/volition.cpp src/intention.cpp)
target_include_directories(exotic_core PUBLIC include)
target_link_libraries(exotic_core PUBLIC exotic_sqlite)
add_executable(exotic_core_demo app/main.cpp)
target_link_libraries(exotic_core_demo PRIVATE exotic_core)
add_executable(exotic_free_agent_server app/free_agent_server.cpp)
target_link_libraries(exotic_free_agent_server PRIVATE exotic_core)
if(WIN32)
  target_link_libraries(exotic_free_agent_server PRIVATE ws2_32)
endif()
include(CTest)
if(BUILD_TESTING)
  foreach(test_name IN ITEMS runtime free_agent free_agent_service agent_store self_model volition intention)
    if(test_name STREQUAL "runtime")
      add_executable(exotic_core_tests tests/runtime_tests.cpp)
      target_link_libraries(exotic_core_tests PRIVATE exotic_core)
      add_test(NAME exotic_core_tests COMMAND exotic_core_tests)
    else()
      add_executable(exotic_${test_name}_tests tests/${test_name}_tests.cpp)
      target_link_libraries(exotic_${test_name}_tests PRIVATE exotic_core)
      add_test(NAME exotic_${test_name}_tests COMMAND exotic_${test_name}_tests)
    endif()
  endforeach()
endif()
